cmake_minimum_required(VERSION 3.25)
project(Quadeplus VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QUADEPLUS_STORAGE "Build Arrow/Parquet storage tests and demonstration" OFF)
option(QUADEPLUS_DATABENTO "Build the optional Databento SDK adapter tests" OFF)
option(QUADEPLUS_DATABENTO_SMOKE "Build the credentialed, billable provider smoke executable" OFF)
option(QUADEPLUS_SANITIZERS "Instrument project code with ASan/UBSan" OFF)
option(QUADEPLUS_BENCHMARKS "Build the data-model benchmark" OFF)

if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(QUADEPLUS_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_library(quadeplus_core INTERFACE)
add_library(Quadeplus::core ALIAS quadeplus_core)
target_include_directories(quadeplus_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(quadeplus_core INTERFACE cxx_std_20)

enable_testing()
foreach(TEST IN ITEMS data_model_tests data_quality_tests ingestion_query_tests
    ingestion_ports_tests ingestion_result_tests ingestion_provider_tests
    ingestion_query_plan_tests synchronous_ingestor_tests data_integrity_tests
    ingestion_retention_tests)
  add_executable(${TEST} "tests/${TEST}.cpp")
  target_link_libraries(${TEST} PRIVATE Quadeplus::core)
  target_compile_options(${TEST} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ${TEST} COMMAND ${TEST})
endforeach()

if(QUADEPLUS_STORAGE)
  add_subdirectory(integration/storage)
endif()
if(QUADEPLUS_DATABENTO_SMOKE AND NOT QUADEPLUS_DATABENTO)
  message(FATAL_ERROR "QUADEPLUS_DATABENTO_SMOKE requires QUADEPLUS_DATABENTO=ON")
endif()
if(QUADEPLUS_DATABENTO)
  add_subdirectory(integration/databento)
endif()
if(QUADEPLUS_BENCHMARKS)
  add_executable(quadeplus_data_model_benchmark benchmarks/data_model_benchmark.cpp)
  target_link_libraries(quadeplus_data_model_benchmark PRIVATE Quadeplus::core)
  target_compile_options(quadeplus_data_model_benchmark PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
